    72→    72→export function resolveBones(clip) {
    73→    73→  const found = {};
    74→    74→  // 1) Alias-Tabelle
    75→    75→  for (const role of ROLE_ORDER) {
    76→    76→    for (const a of BONE_ALIASES[role]) {
    77→    77→      const idx = clip.bestNodeFor(a);
    78→    78→      if (idx !== undefined) { found[role] = idx; break; }
    79→    79→    }
    80→    80→  }
    81→    81→  // 2) Heuristik für noch Fehlende — animierte Knoten bevorzugen
    82→    82→  const used = new Set(Object.values(found));
    83→    83→  for (const [role, re, sideRe] of HEURISTICS) {
    84→    84→    if (found[role] !== undefined) continue;
    85→    85→    let best = undefined, bestScore = -1;
    86→    86→    for (let i = 0; i < clip.nodes.length; i++) {
    87→    87→      const n = clip.nodes[i];
    88→    88→      if (!n.name) continue;
    89→    89→      const low = n.name.toLowerCase();
    90→    90→      if (BAD_NAME.test(low)) continue;
    91→    91→      if (!re.test(low)) continue;
    92→    92→      if (sideRe && !sideRe.test(low)) continue;
    93→    93→      if (used.has(i)) continue;
    94→    94→      // 'forearm' darf nicht 'arm' klauen — HEURISTICS-Reihenfolge regelt das
    95→    95→      const animated = clip.rotationTracks.has(i) ? 1 : 0;
    96→    96→      const score = animated * 2 + (n.name.length < 24 ? 1 : 0);
    97→    97→      if (score > bestScore) { bestScore = score; best = i; }
    98→    98→    }
    99→    99→    if (best !== undefined) { found[role] = best; used.add(best); }
   100→   100→  }
   101→   101→  return found;
   102→   102→}

   104→   104→// --- Quaternion-Werkzeuge (x,y,z,w) ---
   105→   105→function quatFromAxisAngle(axis, ang, out) {
   106→   106→  const s = Math.sin(ang / 2);
   107→   107→  out[0] = axis[0] * s; out[1] = axis[1] * s; out[2] = axis[2] * s; out[3] = Math.cos(ang / 2);
   108→   108→  return out;
   109→   109→}
   110→   110→function quatConj(q, out) { out[0] = -q[0]; out[1] = -q[1]; out[2] = -q[2]; out[3] = q[3]; return out; }
   111→   111→function quatMul2(a, b, out) {
   112→   112→  const ax = a[0], ay = a[1], az = a[2], aw = a[3];
   113→   113→  const bx = b[0], by = b[1], bz = b[2], bw = b[3];
   114→   114→  out[0] = aw * bx + ax * bw + ay * bz - az * by;
   115→   115→  out[1] = aw * by - ax * bz + ay * bw + az * bx;
   116→   116→  out[2] = aw * bz + ax * by - ay * bx + az * bw;
   117→   117→  out[3] = aw * bw - ax * bx - ay * by - az * bz;
   118→   118→  return out;
   119→   119→}
   120→   120→function quatLogAxis(q, out) {
   121→   121→  // Axis-Angle aus Quaternion
   122→   122→  const w = Math.min(1, Math.max(-1, q[3]));
   123→   123→  const ang = 2 * Math.acos(w);
   124→   124→  const s = Math.sqrt(Math.max(1e-12, 1 - w * w));
   125→   125→  out[0] = q[0] / s; out[1] = q[1] / s; out[2] = q[2] / s;
   126→   126→  return ang;
   127→   127→}
   128→   128→function quatFromTwoVecs(a, b, out) {
   129→   129→  // Minimale Rotation, die Richtung a → Richtung b dreht (twist-frei).
   130→   130→  const cx = a[1] * b[2] - a[2] * b[1], cy = a[2] * b[0] - a[0] * b[2], cz = a[0] * b[1] - a[1] * b[0];
   131→   131→  const cl = Math.hypot(cx, cy, cz);
   132→   132→  const d = Math.max(-1, Math.min(1, a[0] * b[0] + a[1] * b[1] + a[2] * b[2]));
   133→   133→  const s = Math.sqrt(Math.max(0, 0.5 * (1 + d))); // cos(ang/2)
   134→   134→  if (cl < 1e-9 || s < 1e-6) {
   135→   135→    if (d > 0) { out[0] = 0; out[1] = 0; out[2] = 0; out[3] = 1; return out; }
   136→   136→    // 180°: beliebige senkrechte Achse
   137→   137→    let px = 1, py = 0, pz = 0;
   138→   138→    if (Math.abs(a[0]) > 0.9) { px = 0; py = 1; }
   139→   139→    const ax = a[1] * pz - a[2] * py, ay = a[2] * px - a[0] * pz, az = a[0] * py - a[1] * px;
   140→   140→    const l = Math.hypot(ax, ay, az) || 1;
   141→   141→    out[0] = ax / l; out[1] = ay / l; out[2] = az / l; out[3] = 0;
   142→   142→    return out;
   143→   143→  }
   144→   144→  out[0] = cx / (2 * s); out[1] = cy / (2 * s); out[2] = cz / (2 * s); out[3] = s;
   145→   145→  const n = Math.hypot(out[0], out[1], out[2], out[3]) || 1;
   146→   146→  out[0] /= n; out[1] /= n; out[2] /= n; out[3] /= n;
   147→   147→  return out;
   148→   148→}
   149→   149→function rotVecByQuat(q, v, out) {
   150→   150→  // v' = q ⊗ v ⊗ q* (Standard-Sandwich, KONJUGIERTER zweiter Faktor —
   151→   151→  // die alte Version multiplizierte mit q selbst → gespiegelte Achsen)
   152→   152→  const x = q[0], y = q[1], z = q[2], w = q[3];
   153→   153→  const vx = v[0], vy = v[1], vz = v[2];
   154→   154→  const tx = 2 * (y * vz - z * vy);
   155→   155→  const ty = 2 * (z * vx - x * vz);
   156→   156→  const tz = 2 * (x * vy - y * vx);
   157→   157→  out[0] = vx + w * tx + (y * tz - z * ty);
   158→   158→  out[1] = vy + w * ty + (z * tx - x * tz);
   159→   159→  out[2] = vz + w * tz + (x * ty - y * tx);
   160→   160→  return out;
   161→   161→}

   163→   163→// Welt-Achse eines MuJoCo-Gelenks (im Keyframe, d. h. Ruhepose)
   164→   164→function jointWorldAxis(sim, jid, out) {
   165→   165→  const mod = sim.model;
   166→   166→  const bid = mod.jnt_bodyid[jid];
   167→   167→  // jnt_axis ist im Rahmen des Kind-Körpers lokal
   168→   168→  const ax = [mod.jnt_axis[3 * jid], mod.jnt_axis[3 * jid + 1], mod.jnt_axis[3 * jid + 2]];
   169→   169→  // MuJoCo xquat ist (w,x,y,z) → in [x,y,z,w] drehen für rotVecByQuat!
   170→   170→  const q = [sim._xquat[4 * bid + 1], sim._xquat[4 * bid + 2], sim._xquat[4 * bid + 3], sim._xquat[4 * bid]];
   171→   171→  rotVecByQuat(q, ax, out);
   172→   172→  return out;
   173→   173→}

   182→   182→export function retargetToG1(clip, sim, log = () => {}) {
   183→   183→  const bones = resolveBones(clip);
   184→   184→  const need = ['hips', 'leftUpLeg', 'leftLeg', 'rightUpLeg', 'rightLeg'];

   192→   192→
   193→   193→  // Namen der gelösten Knochen für sampleWorld (normalisiert kompatibel)
   194→   194→  const wanted = roleNames.map(r => clip.nodes[bones[r]].name);
   195→   195→  // Lehrer-Ghost: Rollen in fester Reihenfolge (nur vorhandene)
   196→   196→  const ghostRoles = GHOST_ROLES.filter(r => bones[r] !== undefined);
   197→   197→
   198→   198→  const nu = sim.nu;
   199→   199→  const A = sim.actByName;
   200→   200→  // Gelenk-Weltachsen im Keyframe sammeln
   201→   201→  const axisOf = {};
   202→   202→  for (const name of sim.actName) {
   203→   203→    const aid = A[name];
   204→   204→    const jid = sim.actJoint[aid];
   205→   205→    axisOf[name] = jointWorldAxis(sim, jid, [0, 0, 0]);
   206→   206→  }
   207→   207→  const clampA = (name, v) => {
   208→   208→    const a = A[name];
   209→   209→    return Math.min(sim.actRange[2 * a + 1], Math.max(sim.actRange[2 * a], v));
   210→   210→  };
   211→   211→
   212→   212→  // Achsen-Mapping GLB (Y-hoch, +Z-Blick) → MuJoCo (Z-hoch, +X-Blick):
   213→   213→  // X_glb→Y_mjc (lateral), Y_glb→Z_mjc (hoch), Z_glb→X_mjc (vorwärts)
   214→   214→  // = Rotation 120° um die Raumdiagonale (1,1,1): Quaternion [0.5,0.5,0.5,0.5]
   215→   215→  const ALIGN = [0.5, 0.5, 0.5, 0.5];
   216→   216→  const ALIGN_INV = [-0.5, -0.5, -0.5, 0.5];
   217→   217→
   218→   218→  // „Brust“ = nächster Vorfahre der Arme — deren WELTrotation enthält die
   219→   219→  // komplette Oberkörperbeugung (Wirbelsäulen-Kette akkumuliert). Die G1-
   220→   220→  // Basis trägt diese Orientierung (inkl. Ruhepose-Beugung wie beim Zombie).
   221→   221→  const armNode = bones.leftArm !== undefined ? bones.leftArm
   222→   222→    : (bones.rightArm !== undefined ? bones.rightArm : undefined);
   223→   223→  let chestIdx = armNode !== undefined ? clip.parentOf.get(armNode) : undefined;
   224→   224→  if (chestIdx === undefined || chestIdx === bones.hips) chestIdx = bones.spine;
   225→   225→  if (chestIdx === undefined) chestIdx = bones.hips;
   226→   226→
   227→   227→  const fps = Math.min(30, Math.max(15, clip.fpsHint));
   228→   228→  const n = Math.max(2, Math.round(clip.duration * fps));
   229→   229→  const q = new Float32Array(n * nu);
   230→   230→  const h = new Float32Array(n);
   231→   231→
   232→   232→  const worldMap = new Map();
   233→   233→  const conjTmp = [0, 0, 0, 1];
   234→   234→  const axisTmp = [0, 0, 0];
   235→   235→  const qTmp = [0, 0, 0, 1];
   236→   236→  const qTmp2 = [0, 0, 0, 1];
   237→   237→  const qId = [0, 0, 0, 1];
   238→   238→  const tA = [0, 0, 0, 1], tB = [0, 0, 0, 1];
   239→   239→
   240→   240→  // ── Ruhewelt-Rotationen der Quell-Knochen (Kalibrierungs-Referenz) ──
   241→   241→  // WICHTIG: auch matrix-basierte Nodes (assimp _$AssimpFbx$_ PreRotation!)
   242→   242→  // einbeziehen, sonst ist die Ruhelage falsch.
   243→   243→  const localRestRot = (idx) => {
   244→   244→    const n = clip.nodes[idx];
   245→   245→    if (n.rotation) return n.rotation.slice();
   246→   246→    if (n.matrix) return clip._decompMatrix(idx).q.slice();
   247→   247→    return [0, 0, 0, 1];
   248→   248→  };
   249→   249→  const restQ = new Map();
   250→   250→  const computeRest = (idx) => {
   251→   251→    if (restQ.has(idx)) return restQ.get(idx);
   252→   252→    const local = localRestRot(idx);
   253→   253→    const p = clip.parentOf.get(idx);
   254→   254→    const w = p === undefined ? local : quatMul2(computeRest(p), local, [0, 0, 0, 1]);
   255→   255→    restQ.set(idx, w);
   256→   256→    return w;
   257→   257→  };
   258→   258→  const calRoles = ['hips', 'spine', 'leftUpLeg', 'leftLeg', 'leftFoot', 'rightUpLeg', 'rightLeg', 'rightFoot', 'leftArm', 'leftForeArm', 'rightArm', 'rightForeArm'];
   259→   259→  for (const r of calRoles) if (bones[r] !== undefined) computeRest(bones[r]);
   260→   260→  computeRest(chestIdx);
   261→   261→
   262→   262→  // ── G1-KALIBRIERUNG: Körper-Orientierungen in der NULL-Pose ──
   263→   263→  // Knochen-Frames verschiedener Rigs sind beliebig konventioniert — direkt
   264→   264→  // verglichen werden dürfen nur die jeweiligen RUHELAGEN. CAL[seite][glied]
   265→   265→  // = (alignierte Quell-Ruhe-Weltrotation)⁻¹ ⊗ G1-Körper-Nullpose-Quat.
   266→   266→  // Damit gilt: Quell-Ruhepose ↔ G1-Nullpose EXAKT, Animation relativ dazu.
   267→   267→  const bodyOfAct = (name) => sim.model.jnt_bodyid[sim.actJoint[sim.actByName[name]]];
   268→   268→  const ghost0 = sim.makeGhostData();
   269→   269→  sim._mjApi.mj_resetData(sim.model, ghost0);
   270→   270→  sim._mjApi.mj_forward(sim.model, ghost0);
   271→   271→  // MuJoCo xquat ist (w,x,y,z) → in [x,y,z,w] drehen (Projekt-Konvention)
   272→   272→  const zeroQuat = (b) => [ghost0.xquat[4 * b + 1], ghost0.xquat[4 * b + 2], ghost0.xquat[4 * b + 3], ghost0.xquat[4 * b]];
   273→   273→  const makeCal = (srcIdx, g1Body) => {
   274→   274→    const ar = quatMul2(ALIGN, computeRest(srcIdx), [0, 0, 0, 1]);
   275→   275→    const aligned = quatMul2(ar, ALIGN_INV, [0, 0, 0, 1]);
   276→   276→    return quatMul2(quatConj(aligned, [0, 0, 0, 1]), zeroQuat(g1Body), [0, 0, 0, 1]);
   277→   277→  };
   278→   278→  const CAL = {};
   279→   279→  for (const side of ['left', 'right']) {

   306→   306→  //

   308→   308→  const pos0Of = (b) => [ghost0.xpos[3 * b], ghost0.xpos[3 * b + 1], ghost0.xpos[3 * b + 2]];

   362→   362→  }

   364→   364→  for (const side of ['left', 'right']) {

   366→   366→      const e = limbEnd[side][k];

   371→   371→  // Slots: 0 OS-L, 1 SB-L, 2 OA-L, 3 UA-L, 4 OS-R, 5 SB-R, 6 OA-R, 7 UA-R
   372→   372→  const dSrc = new Float32Array(n * 8 * 3).fill(NaN);
   373→   373→  // alignierte Lehrer-Fuß-Weltquats (L/R) für die Knöchel-Orientierung
   374→   374→  const footW = new Float32Array(n * 2 * 4).fill(NaN);
   375→   375→
   376→   376→  const triadQ = [0, 0, 0, 1]; // wird je Frame aus KnochenPOSITIONEN gebaut
   377→   377→  // Einachser (nur noch Taille): Projektion des Ziel-Achsenwinkels auf die Gelenkachse
   378→   378→  const project1 = (jointName, dq) => {
   379→   379→    const ang = quatLogAxis(dq, axisTmp);
   380→   380→    const a = axisOf[jointName];
   381→   381→    return clampA(jointName, ang * (axisTmp[0] * a[0] + axisTmp[1] * a[1] + axisTmp[2] * a[2]));

   383→   383→
   384→   384→  let hipsRestH = 0;

   395→   395→  const g1StandH = sim._xpos[3 * sim.baseBody + 2];
   396→   396→  const hScale = g1StandH / Math.max(0.2, hipsRestH * scale);
   397→   397→
   398→   398→  const rawH = new Float32Array(n);
   399→   399→  // Triaden-Yaw je Frame (dieselbe Größe, die baseQ herausdreht — die
   400→   400→  // IK-Ziele werden damit KONJUGIERT, sodass Lehrer-Ziele und Geist-Basis
   401→   401→  // garantiert im selben ent-yawten Rahmen liegen)

   403→   403→  // Basis-Orientierung je Frame (Brust-Weltrotation, ALIGN-konjugiert,
   404→   404→  // Yaw-Anteil entfernt — die Blickrichtung besitzt der Root-Motion-Yaw).
   405→   405→  // Trägt die ABSOLUTE Oberkörper-Pose (auch Ruhepose-Beugung wie beim
   406→   406→  // Zombie) — die Hüften kompensieren, damit die Füße unten bleiben.
   407→   407→  const baseQ = new Float32Array(n * 4);
   408→   408→  // Root-Motion: Bahn des Hüftpunkts (MuJoCo-Rahmen, relativ zu Frame 0)
   409→   409→  // + Blickrichtung (Yaw) — der Referenz-Geist läuft damit WIRKLICH durchs Feld.
   410→   410→  const root = new Float32Array(n * 2);
   411→   411→  const yaw = new Float32Array(n);
   412→   412→  const rawYaw = new Float32Array(n);
   413→   413→  // Lehrer-Ghost: Weltpositionen aller Rollenknochen je Frame (MuJoCo-Rahmen, m)
   414→   414→  const srcPos = new Float32Array(n * ghostRoles.length * 3);
   415→   415→  const srcJoints = ghostRoles.slice();
   416→   416→  const worldPos = new Map();
   417→   417→  const FWD_GLB = [0, 0, 1]; // GLB: +Z ist Blickrichtung (Annahme wie Achsen-Mapping)
   418→   418→  const fwdTmp = [0, 0, 0];
   419→   419→  const alignedQ = [0, 0, 0, 1];
   420→   420→  let sx0 = 0, sy0 = 0;
   421→   421→  for (let f = 0; f < n; f++) {
   422→   422→    const t = f / fps;
   423→   423→    clip.sampleWorldFull(t, wanted, worldMap, worldPos);
   424→   424→    const off = f * nu;
   425→   425→    // ── Basis-Triade aus KnochenPOSITIONEN (konventionsfrei) ──
   426→   426→    // up  = Richtung Hüfte→Brust — die VISUELLE Torso-Achse (erfasst die
   427→   427→    //       komplette Beugung, auch eine in der Ruhepose steckende)
   428→   428→    // fwd = Blick der Hüfte (GLB +Z rotiert mit W_hips), orthogonal auf up
   429→   429→    {
   430→   430→      const hp3 = worldPos.get(bones.hips), cp3 = worldPos.get(chestIdx) || hp3;
   431→   431→      let ux = 0, uy = 0, uz = 1; // MJC hoch (Fallback aufrecht)
   432→   432→      if (hp3 && cp3) {
   433→   433→        const g = rotVec(ALIGN, [cp3[0] - hp3[0], cp3[1] - hp3[1], cp3[2] - hp3[2]], [0, 0, 0]);

   435→   435→        if (len > 1e-9) { ux = g[0] / len; uy = g[1] / len; uz = g[2] / len; }

   438→   438→      const fG = rotVec(ALIGN, rotVec(wHipsQ, FWD_GLB, [0, 0, 0]), [0, 0, 0]);

   442→   442→      if (fl < 1e-6) { fx = 1 - ux * ux; fy = -ux * uy; fz = -ux * uz; fl = Math.hypot(fx, fy, fz) || 1; }

   447→   447→    // Taille: Verdrehung Becken↔Brust (Yaw-Anteil) — die BEUGUNG selbst
   448→   448→    // trägt die Basis-Orientierung, nicht die Taille (G1 hat nur waist_yaw)
   449→   449→    quatConj(worldMap.get(bones.hips) || qId, conjTmp);
   450→   450→    quatMul2(conjTmp, worldMap.get(chestIdx) || qId, alignedQ);
   451→   451→    quatMul2(ALIGN, alignedQ, qTmp); quatMul2(qTmp, ALIGN_INV, alignedQ);

   453→   453→    q[off + A[waistName]] = project1(waistName, alignedQ);
   454→   454→    // ── IK-Ziele sammeln: Lehrer-Richtungen (Positions-Differenzen sind
   455→   455→    // konventionsfrei!) + Fuß-Quats; die Gelenke selbst löst Pass 2 ──
   456→   456→    const storeDir = (aIdx, bIdx, slot) => {
   457→   457→      const a = worldPos.get(aIdx), b = worldPos.get(bIdx);
   458→   458→      if (!a || !b) return;
   459→   459→      const g = rotVec(ALIGN, [b[0] - a[0], b[1] - a[1], b[2] - a[2]], [0, 0, 0]);
   460→   460→      const l = Math.hypot(g[0], g[1], g[2]);
   461→   461→      if (l < 1e-9) return;
   462→   462→      const o = (f * 8 + slot) * 3;
   463→   463→      dSrc[o] = g[0] / l; dSrc[o + 1] = g[1] / l; dSrc[o + 2] = g[2] / l;
   464→   464→    };

   468→   468→      // Ganze-Bein-Richtung nach hinten, obwohl der Oberschenkel nach vorn
   469→   469→      // zeigt — der 27-42°-„Restfehler" war genau dieser Messfehler)

   471→   471→      storeDir(bones[side + 'Leg'], limbEnd[side].shin, li + 1);
   472→   472→      if (bones[side + 'Arm'] !== undefined && bones[side + 'ForeArm'] !== undefined) {
   473→   473→        storeDir(bones[side + 'Arm'], bones[side + 'ForeArm'], li + 2);
   474→   474→        if (limbEnd[side].fore !== undefined) storeDir(bones[side + 'ForeArm'], limbEnd[side].fore, li + 3);
   475→   475→      }
   476→   476→      if (bones[side + 'Foot'] !== undefined) {
   477→   477→        alignedOf(bones[side + 'Foot'], tA);

   488→   488→      triadYaw[f] = yawC;
   489→   489→      const cy2 = Math.cos(-yawC / 2), sy2 = Math.sin(-yawC / 2);
   490→   490→      baseQ[4 * f]     = cy2 * qx + sy2 * qy;
   491→   491→      baseQ[4 * f + 1] = cy2 * qy - sy2 * qx;
   492→   492→      baseQ[4 * f + 2] = cy2 * qz - sy2 * qw;
   493→   493→      baseQ[4 * f + 3] = cy2 * qw + sy2 * qz;
   494→   494→    }

   514→   514→  // stabile Blick — die Hüft-Vorwärtsachse kippt beim Laufen stark mit
   515→   515→  // (Beckenrotation/Neigung). Bei Stillstand (Idle) liefert die Hüft-
   516→   516→  // Rotation die Blickrichtung. Danach unwrap + glätten + relativ zu Frame 0.

   520→   520→    let lastY = null;
   521→   521→    for (let f = 0; f < n; f++) {

   523→   523→      const dx = root[2 * p1] - root[2 * p0], dy = root[2 * p1 + 1] - root[2 * p0 + 1];
   524→   524→      const dt = Math.max(1e-6, (p1 - p0) / fps);
   525→   525→      if (Math.hypot(dx, dy) / dt > SPEED_MIN) { lastY = Math.atan2(dy, dx); dirYaw[f] = lastY; }
   526→   526→      else dirYaw[f] = lastY !== null ? lastY : rawYaw[f];
   527→   527→    }
   528→   528→    for (let f = 1; f < n; f++) {
   529→   529→      let d = dirYaw[f] - dirYaw[f - 1];
   530→   530→      while (d > Math.PI) d -= 2 * Math.PI;
   531→   531→      while (d < -Math.PI) d += 2 * Math.PI;
   532→   532→      dirYaw[f] = dirYaw[f - 1] + d;
   533→   533→    }
   534→   534→    smooth(dirYaw, 5);
   535→   535→    const y0 = dirYaw[0];
   536→   536→    for (let f = 0; f < n; f++) {
   537→   537→      let d = dirYaw[f] - y0;
   538→   538→      while (d > Math.PI) d -= 2 * Math.PI;
   539→   539→      while (d < -Math.PI) d += 2 * Math.PI;
   540→   540→      yaw[f] = d;
   541→   541→    }

   552→   552→  // lief bei ~100°-Startfehlern in Verdreh-Täler: Hüften liefen ins Yaw/Roll-

   556→   556→    // so liegen Lehrer-Limb-Ziele und Geist-Basis garantiert im selben Rahmen.

   565→   565→      return out;

   573→   573→    };

   856→   856→    sim.setGhostPose(ghost, q, off, rawH[f], 0, 0, 0, bq4);
   857→   857→    let lowest = 0;
   858→   858→    // Füße: unterster Punkt aller Fuß-Geoms (Welt-z = Körper-hoch + Tiefen-Offset)
   859→   859→    for (const fg of footGeoms) {
   860→   860→      const z = ghost.xpos[3 * fg.body + 2] + fg.lowZ;
   861→   861→      if (z < lowest) lowest = z;
   862→   862→    }
   863→   863→    h[f] = Math.min(1.15, Math.max(0.4, rawH[f] - lowest));
   864→   864→  }
   865→   865→  // Höhe glätten (Burgen vermeiden)…
   866→   866→  smooth(h, 5);
   867→   867→  // …und DANACH exakt erden: mit der GEGLÄTTETEN Höhe erneut messen und den
   868→   868→  // Rest-Auftrieb abziehen (das Glätten hebt den tiefsten Fuß sonst cm-weise an).
   869→   869→  // Echtes Minimum (auch positiv!) — sonst bleibt ein cm-Hover stehen.
   870→   870→  for (let f = 0; f < n; f++) {
   871→   871→    const off = f * nu;
   872→   872→    bq4[0] = baseQ[4 * f]; bq4[1] = baseQ[4 * f + 1]; bq4[2] = baseQ[4 * f + 2]; bq4[3] = baseQ[4 * f + 3];
   873→   873→    sim.setGhostPose(ghost, q, off, h[f], 0, 0, 0, bq4);
   874→   874→    let lowest = Infinity;
   875→   875→    for (const fg of footGeoms) {
   876→   876→      const z = ghost.xpos[3 * fg.body + 2] + fg.lowZ;
   877→   877→      if (z < lowest) lowest = z;
   878→   878→    }
   879→   879→    if (!Number.isFinite(lowest)) lowest = 0;
   880→   880→    h[f] = Math.min(1.15, Math.max(0.4, h[f] - lowest));
   881→   881→  }
   882→   882→
   883→   883→  return {
   884→   884→    name: clip.name || 'clip',
   885→   885→    fps, n, nu,
   886→   886→    q, h,
   887→   887→    root, yaw, srcPos, srcJoints,
   888→   888→    baseQ, // Basis-Orientierung je Frame (n×4, xyzw) — Lehrer-Nick/Roll für den Geist
   889→   889→    scale, // Datei-Einheit → Meter (für den Original-Mesh-Wrap in render3d)
   890→   890→    mergedFrom: clip.mergedFrom || 0,
   891→   891→    mapped: roleNames,
   892→   892→    duration: clip.duration,
   893→   893→  };
   894→   894→}

   896→   896→// Fuß-Geoms mit ECHTER Tiefe: unterster Punkt je Geom im KÖRPER-Frame
   897→   897→// (Meshes: unterste Vertebra inkl. geom_quat; Kugel/Kapsel/Zylinder/Box: Formel)
   898→   898→function findFootGeoms(sim) {
   899→   899→  const out = [];
   900→   900→  const m = sim.model;
   901→   901→  for (let g = 0; g < sim.ngeom; g++) {
   902→   902→    const b = m.geom_bodyid[g];
   903→   903→    const name = mjName(sim, b);
   904→   904→    if (!name || !/ankle|foot/i.test(name)) continue;
   905→   905→    const type = m.geom_type[g];
   906→   906→    const pz = m.geom_pos[3 * g + 2];
   907→   907→    let lowZ = pz;
   908→   908→    if (type === 7) { // Mesh: unterste Vertebra (geom_quat einrechnen)
   909→   909→      const qx = m.geom_quat[4 * g + 1], qy = m.geom_quat[4 * g + 2], qz = m.geom_quat[4 * g + 3], qw = m.geom_quat[4 * g];
   910→   910→      const dId = m.geom_dataid[g], vAdr = m.mesh_vertadr[dId], vNum = m.mesh_vertnum[dId];
   911→   911→      let minVz = 0;
   912→   912→      for (let i = 0; i < vNum; i++) {
   913→   913→        const vx = m.mesh_vert[3 * (vAdr + i)], vy = m.mesh_vert[3 * (vAdr + i) + 1], vz = m.mesh_vert[3 * (vAdr + i) + 2];
   914→   914→        const tx = 2 * (qy * vz - qz * vy), ty = 2 * (qz * vx - qx * vz), tz = 2 * (qx * vy - qy * vx);
   915→   915→        const z = vz + qw * tz + (qx * ty - qy * tx); // z von q⊗v⊗q*
   916→   916→        if (z < minVz) minVz = z;
   917→   917→      }
   918→   918→      lowZ = pz + minVz;
   919→   919→    } else if (type === 2) lowZ = pz - m.geom_size[3 * g];                          // Kugel
   920→   920→    else if (type === 3) lowZ = pz - (m.geom_size[3 * g] + m.geom_size[3 * g + 1]); // Kapsel (Z-Achse)
   921→   921→    else if (type === 5) lowZ = pz - m.geom_size[3 * g + 1];                        // Zylinder
   922→   922→    else if (type === 6) lowZ = pz - m.geom_size[3 * g + 2];                        // Box
   923→   923→    out.push({ body: b, lowZ });
   924→   924→  }
   925→   925→  return out;
   926→   926→}

   928→   928→// Rotationsmatrix (Spalten X=fwd, Y=left, Z=up) → Quaternion [x,y,z,w]
   929→   929→// (Shepperd-Methode — für die konventionsfreie Basis-Triade)
   930→   930→function mat3ToQuat(xx, xy, xz, yx, yy, yz, zx, zy, zz, out) {
   931→   931→  const m00 = xx, m01 = yx, m02 = zx;
   932→   932→  const m10 = xy, m11 = yy, m12 = zy;
   933→   933→  const m20 = xz, m21 = yz, m22 = zz;
   934→   934→  const tr = m00 + m11 + m22;
   935→   935→  let qx = 0, qy = 0, qz = 0, qw = 1, S;
   936→   936→  if (tr > 0) {
   937→   937→    S = Math.sqrt(tr + 1) * 2; qw = 0.25 * S;
   938→   938→    qx = (m21 - m12) / S; qy = (m02 - m20) / S; qz = (m10 - m01) / S;
   939→   939→  } else if (m00 > m11 && m00 > m22) {
   940→   940→    S = Math.sqrt(1 + m00 - m11 - m22) * 2; qw = (m21 - m12) / S; qx = 0.25 * S;
   941→   941→    qy = (m01 + m10) / S; qz = (m02 + m20) / S;
   942→   942→  } else if (m11 > m22) {
   943→   943→    S = Math.sqrt(1 + m11 - m00 - m22) * 2; qw = (m02 - m20) / S; qy = 0.25 * S;
   944→   944→    qx = (m01 + m10) / S; qz = (m12 + m21) / S;
   945→   945→  } else {
   946→   946→    S = Math.sqrt(1 + m22 - m00 - m11) * 2; qw = (m10 - m01) / S; qz = 0.25 * S;
   947→   947→    qx = (m02 + m20) / S; qy = (m12 + m21) / S;
   948→   948→  }
   949→   949→  const n = Math.hypot(qx, qy, qz, qw) || 1;
   950→   950→  out[0] = qx / n; out[1] = qy / n; out[2] = qz / n; out[3] = qw / n;
   951→   951→  return out;
   952→   952→}

   955→   955→}

   957→   957→function smooth(arr, win) {
   958→   958→  const n = arr.length;
   959→   959→  const out = new Float32Array(n);
   960→   960→  for (let i = 0; i < n; i++) {
   961→   961→    let s = 0, c = 0;
   962→   962→    for (let k = -win; k <= win; k++) {
   963→   963→      const j = i + k;
   964→   964→      if (j >= 0 && j < n) { s += arr[j]; c++; }
   965→   965→    }
   966→   966→    out[i] = s / c;
   967→   967→  }
   968→   968→  arr.set(out);
   969→   969→}
