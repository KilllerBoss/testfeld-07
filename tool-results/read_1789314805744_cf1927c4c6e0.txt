    67→    67→async function loadRobot(id, first = false) {
    68→    68→  if (S.switching) return;
    69→    69→  const cfg = getRobot(id);
    70→    70→  if (!cfg) return;
    71→    71→  S.switching = true;

    73→    73→    stopTraining(true);

    78→    78→    const t0 = performance.now();

    80→    80→    const sim = new RobotSim(cfg, cfg.scene);
    81→    81→    const ms = Math.round(performance.now() - t0);
    82→    82→
    83→    83→    // Alte Simulation freigeben
    84→    84→    if (S.sim) {
    85→    85→      const oldDir = S.sim.cfg.dir;
    86→    86→      S.sim.dispose();

    88→    88→      S.sim = null;

    91→    91→    S.robotId = id;
    92→    92→
    93→    93→    // Gang-/Flugregler
    94→    94→    S.gait = cfg.gait(cfg);
    95→    95→    if (cfg.drone && S.gait.init) { S.gait.init(sim); sim.flightCtl = S.gait; }
    96→    96→    // Trainingsaufgabe: GLB-Motion-Tracking (nur G1 mit aktivem Clip) sonst Geschwindigkeit
    97→    97→    if (id === 'g1' && S.motionClip) {
    98→    98→      S.task = makeMotionTask(cfg, S.motionClip, sim);
    99→    99→      S.task.animOn = S.animTraining;
   100→   100→      // Steuerungs-Wahl des aktiven Clips restaurieren (v2.5.0)

   163→   163→    // seit Root-Folgen einen anderen Beobachtungsraum als Speed; v2.5.0

   165→   165→    if (S.task && (p.obsDim !== S.task.obsDim || p.actDim !== S.task.actDim)) {

   187→   187→const DEFAULT_PPO = { T: 1024, gamma: 0.99, lam: 0.95, clip: 0.2, epochs: 4, mb: 256, lr: 3e-4, cV: 0.5, cE: 0.005, maxGrad: 0.5 };

   190→   190→  if (S.task && S.task.kind === 'motion') return 'motion';

   195→   195→function aiCtx() {
   196→   196→  const cfg = S.sim ? S.sim.cfg : null;
   197→   197→  const kind = aiTaskKind();
   198→   198→  const cur = {};
   199→   199→  if (kind === 'speed' && cfg) {
   200→   200→    cur.rW = cfg.rW; cur.cmd = cfg.cmd; cur.done = cfg.done;
   201→   201→    cur.actSpan = cfg.actSpan; cur.speedMax = cfg.speedMax;
   202→   202→  } else if (kind === 'motion') {
   203→   203→    cur.motionR = MOTION_R;
   204→   204→  } else if (kind === 'hover' && cfg) {
   205→   205→    cur.hoverR = HOVER_R; cur.cmd = cfg.cmd;
   206→   206→  }
   207→   207→  cur.ppo = S.trainer
   208→   208→    ? (({ lr, gamma, lam, clip, epochs, mb, T, cV, cE, maxGrad }) => ({ lr, gamma, lam, clip, epochs, mb, T, cV, cE, maxGrad }))(S.trainer.h)
   209→   209→    : Object.assign({}, DEFAULT_PPO, PPO_OVERRIDES);
   210→   210→  return {
   211→   211→    robot: S.robotId,
   212→   212→    robotName: cfg ? cfg.longName : '—',
   213→   213→    taskKind: kind,
   214→   214→    current: cur,
   215→   215→    history: S.aiHistory,
   216→   216→  };
   217→   217→}

   219→   219→function _aiPPOApply(ppo, touched) {
   220→   220→  for (const [k, v] of Object.entries(ppo)) {
   221→   221→    PPO_OVERRIDES[k] = v;
   222→   222→    if (S.trainer && k !== 'T') S.trainer.h[k] = v;
   223→   223→    touched.push('ppo.' + k);
   224→   224→  }
   225→   225→}

   227→   227→// Patch auf die laufende Konfiguration anwenden (cfg-Objekte werden von
   228→   228→// den Aufgaben live gelesen → Wirkung ohne Neustart).
   229→   229→function applyAIPatch(patch, opts = {}) {
   230→   230→  if (!patch || !Object.keys(patch).length) return false;
   231→   231→  const cfg = S.sim ? S.sim.cfg : null;
   232→   232→  const kind = aiTaskKind();
   233→   233→  const touched = [];
   234→   234→  if (patch.push && patch.push.impulse !== undefined) {
   235→   235→    S.pushStrength = savePushStrength(patch.push.impulse);
   236→   236→    touched.push('push.impulse=' + S.pushStrength.toFixed(1));
   237→   237→  }
   238→   238→  if (cfg) {
   239→   239→    if (kind === 'speed') {
   240→   240→      if (patch.rW) for (const [k, v] of Object.entries(patch.rW)) { cfg.rW[k] = v; touched.push('rW.' + k); }
   241→   241→      if (patch.cmd) {
   242→   242→        if (patch.cmd.vx) { cfg.cmd.vx = patch.cmd.vx.slice(); touched.push('cmd.vx=' + patch.cmd.vx.map(x => x.toFixed(2)).join('..')); }
   243→   243→        if (patch.cmd.yaw) { cfg.cmd.yaw = patch.cmd.yaw.slice(); touched.push('cmd.yaw'); }
   244→   244→      }
   245→   245→      if (patch.done) for (const [k, v] of Object.entries(patch.done)) { cfg.done[k] = v; touched.push('done.' + k); }
   246→   246→      if (patch.actSpan !== undefined) { cfg.actSpan = patch.actSpan; touched.push('actSpan'); }
   247→   247→    } else if (kind === 'hover') {
   248→   248→      if (patch.hoverR) for (const [k, v] of Object.entries(patch.hoverR)) { HOVER_R[k] = v; touched.push('hoverR.' + k); }
   249→   249→      if (patch.cmd && patch.cmd.vx) { cfg.cmd.vx = patch.cmd.vx.slice(); touched.push('cmd.vx'); }
   250→   250→    }
   251→   251→  }
   252→   252→  if (patch.motionR) for (const [k, v] of Object.entries(patch.motionR)) { MOTION_R[k] = v; touched.push('motionR.' + k); }
   253→   253→  if (patch.hoverR && kind !== 'hover') for (const [k, v] of Object.entries(patch.hoverR)) { HOVER_R[k] = v; touched.push('hoverR.' + k); }
   254→   254→  if (patch.ppo) _aiPPOApply(patch.ppo, touched);
   255→   255→  saveAICfg();
   256→   256→  log('KI-Anpassung übernommen: ' + touched.join(', '), 'ok');
   257→   257→  ui.toast('KI-Anpassung übernommen' + (opts.resetTraining ? ' — Training zurückgesetzt' : ''));
   258→   258→  if (opts.resetTraining) {
   259→   259→    stopTraining(true);
   260→   260→    S.trainer = null;
   261→   261→    ui.resetRewards();
   262→   262→    S.episodes = 0;
   263→   263→    if (S.sim) S.sim.reset();
   264→   264→    ui.$('tStart').textContent = 'Training starten';
   265→   265→    ui.$('tStart').classList.remove('btn-stop');
   266→   266→    ui.trainStats({ reward: '–', episodes: 0, steps: 0, rate: 0 });
   267→   267→    ui.drawChart();
   268→   268→    log('Training zurückgesetzt (KI-Anpassung war groß — Netz lernt neu)', 'warn');
   269→   269→  }
   270→   270→  return true;
   271→   271→}

   298→   298→function applySavedAICfg(id) {
   299→   299→  const cfg = getRobot(id);
   300→   300→  if (!cfg || cfg.drone) return;
   301→   301→  try {
   302→   302→    const raw = JSON.parse(localStorage.getItem('tr_ai_speed_' + id) || 'null');
   303→   303→    if (!raw) return;
   304→   304→    const v = validatePatch({ rW: raw.rW, cmd: raw.cmd, done: raw.done, actSpan: raw.actSpan });
   305→   305→    if (v.rW) Object.assign(cfg.rW, v.rW);
   306→   306→    if (v.cmd) { if (v.cmd.vx) cfg.cmd.vx = v.cmd.vx; if (v.cmd.yaw) cfg.cmd.yaw = v.cmd.yaw; }
   307→   307→    if (v.done) Object.assign(cfg.done, v.done);
   308→   308→    if (v.actSpan !== undefined) cfg.actSpan = v.actSpan;
   309→   309→  } catch (e) { /* defekt → Standardwerte */ }

   313→   313→function doPush(dir = 'auto', strength = null) {
   314→   314→  if (!S.sim) return 'Kein Roboter geladen';
   315→   315→  const st = strength != null ? Math.min(10, Math.max(0.5, strength)) : S.pushStrength;
   316→   316→  S.sim.pushRandom(dir, st);
   317→   317→  controls.buzz(30);
   318→   318→  ui.toast('SCHUBS! (' + dir + ', Stärke ' + st.toFixed(1) + ')');
   319→   319→  log('Schubs: dir=' + dir + ' Stärke=' + st.toFixed(1) + ' (Δv≈' + st.toFixed(1) + ' m/s)', 'warn');
   320→   320→  return 'Schubs ausgeführt (dir=' + dir + ', strength=' + st.toFixed(1) + ')';
   321→   321→}

   323→   323→function observeState() {
   324→   324→  const sim = S.sim;
   325→   325→  let up = null;
   326→   326→  if (sim) {
   327→   327→    const o = 4 * sim.baseBody;
   328→   328→    const x = sim._xquat[o + 1], y = sim._xquat[o + 2];
   329→   329→    up = 1 - 2 * (x * x + y * y);
   330→   330→  }
   331→   331→  return JSON.stringify({
   332→   332→    robot: S.robotId,
   333→   333→    mode: S.mode,
   334→   334→    training: S.training,
   335→   335→    speed: sim ? +sim.baseSpeed().toFixed(2) : null,
   336→   336→    height: sim ? +sim.baseHeight().toFixed(2) : null,
   337→   337→    upright: up !== null ? +up.toFixed(2) : null,
   338→   338→    episodes: S.episodes,
   339→   339→    policySaved: !!S.trainer,
   340→   340→    motionClips: S.clips.map((c, i) => ({ index: i, name: c.name })),
   341→   341→    activeClip: S.motionClip ? S.motionClip.name : null,
   342→   342→    motionCtrl: S.task && S.task.kind === 'motion'
   343→   343→      ? { ctrlMode: S.task.ctrlMode, animOn: S.task.animOn, cmd: { vx: +S.task.cmd.vx.toFixed(2), wz: +S.task.cmd.wz.toFixed(2) } }
   344→   344→      : null,
   345→   345→    pushStrength: S.pushStrength,
   346→   346→    joystick: controls.joyMap,
   347→   347→    buttons: S.aiButtons.map(b => ({ id: b.id, label: b.label, action: b.action })),
   348→   348→  });
   349→   349→}

   351→   351→function executeAction(action, depth = 0) {
   352→   352→  const act = action;
   353→   353→  if (!act || typeof act !== 'object') return 'Ungültige Aktion';
   354→   354→  switch (act.type) {
   355→   355→    case 'reset':
   356→   356→      resetRobot();
   357→   357→      return 'Roboter zurückgesetzt';
   358→   358→    case 'push':
   359→   359→      return doPush(act.dir || 'auto', act.strength != null ? act.strength : null);
   360→   360→    case 'cmd': {
   361→   361→      if (!S.sim) return 'Kein Roboter geladen';
   362→   362→      S.cruise = { vx: act.vx, yaw: act.yaw, until: performance.now() + act.ms };
   363→   363→      return 'Autofahrt gestartet: vx=' + act.vx.toFixed(2) + ' m/s, yaw=' + act.yaw.toFixed(2) + ' rad/s für ' + Math.round(act.ms / 1000) + ' s (endet früher bei Stick-Bewegung)';
   364→   364→    }
   365→   365→    case 'mode': {
   366→   366→      if (act.mode === 'policy' && !S.trainer) return 'Keine Policy vorhanden — erst Training starten';
   367→   367→      S.mode = act.mode;
   368→   368→      ui.setMode(S.mode);
   369→   369→      if (S.sim) S.sim.reset();
   370→   370→      return 'Modus: ' + act.mode.toUpperCase();
   371→   371→    }
   372→   372→    case 'clip': {
   373→   373→      const rec = S.clips[act.index];
   374→   374→      if (!rec) return 'Kein Clip an Index ' + act.index + ' (vorhanden: ' + S.clips.length + ')';
   375→   375→      activateClip(rec).catch(err => log('Clip-Aktivierung fehlgeschlagen: ' + err.message, 'err'));
   376→   376→      return 'Clip-Aktivierung gestartet: ' + (rec.name || act.index);
   377→   377→    }
   378→   378→    case 'macro': {
   379→   379→      if (depth > 2) return 'Makro-Verschachtelung zu tief';
   380→   380→      let t = 0;
   381→   381→      const results = [];
   382→   382→      for (const step of act.steps.slice(0, 6)) {
   383→   383→        if (step.waitMs !== undefined) { t += step.waitMs; continue; }
   384→   384→        const wait = t;
   385→   385→        const sub = step;
   386→   386→        setTimeout(() => executeAction(sub, depth + 1), wait);
   387→   387→        results.push(sub.type + '@' + wait + 'ms');
   388→   388→      }
   389→   389→      return 'Makro geplant: ' + (results.join(', ') || 'nur Wartezeiten') + ' (Gesamt ' + t + ' ms)';
   390→   390→    }
   391→   391→    default:
   392→   392→      return 'Unbekannter Aktionstyp: ' + (act.type || '?');
   393→   393→  }
   394→   394→}

   397→   397→function execTool(tool, args) {
   398→   398→  try {
   399→   399→    if (tool === 'observe') return observeState();
   400→   400→    if (tool === 'applyConfig') {
   401→   401→      const ok = applyAIPatch(args.patch, { resetTraining: args.resetTraining });
   402→   402→      return ok ? 'Konfiguration angewendet: ' + JSON.stringify(args.patch) : 'Patch war leer — nichts geändert';
   403→   403→    }
   404→   404→    if (tool === 'addButton') {
   405→   405→      const r = addButton({ label: args.label, action: args.action });
   406→   406→      if (!r.ok) return 'Fehler: ' + r.reason;
   407→   407→      S.aiButtons = r.list;
   408→   408→      renderAIButtons();
   409→   409→      return 'Button erstellt: id=' + r.button.id + ' label="' + r.button.label + '" action=' + JSON.stringify(r.button.action);
   410→   410→    }
   411→   411→    if (tool === 'removeButton') {
   412→   412→      const r = removeButton(args.id);
   413→   413→      if (!r.ok) return 'Fehler: ' + r.reason;
   414→   414→      S.aiButtons = r.list;
   415→   415→      renderAIButtons();
   416→   416→      return 'Button entfernt: ' + args.id;
   417→   417→    }
   418→   418→    if (tool === 'mapJoystick') {
   419→   419→      controls.joyMap = validateJoyMap(args);
   420→   420→      saveJoyMap(controls.joyMap);
   421→   421→      return 'Joystick-Map gesetzt: ' + JSON.stringify(controls.joyMap);
   422→   422→    }
   423→   423→    return 'Unbekanntes Werkzeug: ' + tool;
   424→   424→  } catch (e) {
   425→   425→    return 'Werkzeug-Fehler: ' + e.message;
   426→   426→  }
   427→   427→}

   651→   651→// Ein Trainingsschritt (auf Regelrate CTRL_DT)
   652→   652→function trainCtrlStep() {
   653→   653→  const sim = S.sim, task = S.task, trainer = S.trainer;
   654→   654→  const substeps = Math.max(1, Math.round(CTRL_DT / sim.timestep));
   655→   655→  if (task.stepsLeft <= 0) task.sampleCmd(trainer.rng);
   656→   656→
   657→   657→  const o = task.observe(sim, S.obsBuf);
   658→   658→  if (!finiteArr(S.obsBuf)) { sim.reset(); task.reset(trainer.rng, sim); return; }
   659→   659→  trainer.norm.update(S.obsBuf);
   660→   660→  const { act, logp, value } = trainer.act(S.obsBuf, false);
   661→   661→  task.actionToCtrl(sim, act);
   662→   662→  sim.stepN(substeps);
   663→   663→  if (task.kind === 'motion') task.advance(CTRL_DT);
   664→   664→  const { r, done } = task.reward(sim);
   665→   665→  for (let i = 0; i < act.length; i++) task.lastAct[i] = act[i];
   666→   666→  S.epReward += r;
   667→   667→
   668→   668→  const full = trainer.store(S.obsBuf, act, logp, r, done, value);
   669→   669→  if (done) {
   670→   670→    S.episodes++;
   671→   671→    ui.pushEpisodeReward(S.epReward);
   672→   672→    S.epReward = 0;
   673→   673→    sim.reset();
   674→   674→    task.reset(trainer.rng, sim);
   675→   675→  }
   676→   676→  if (full) {
   677→   677→    const lastObs = task.observe(sim, S.obsBuf);
   678→   678→    const lastVal = trainer.act(S.obsBuf, true).value;
   679→   679→    const m = trainer.finishAndUpdate(lastVal);
   680→   680→    m._lastMetrics = m;
   681→   681→  }
   682→   682→}

   685→   685→function applyGait(dtCtrl) {
   686→   686→  const sim = S.sim, cfg = S.sim.cfg;
   687→   687→  dtCtrl = cfg.ctrlDt || dtCtrl;
   688→   688→  // KI-Autofahrt (Cruise): fester Fahrbefehl, endet an Zeitgrenze oder
   689→   689→  // sobald der Nutzer den Stick bewegt (Eingriff geht immer vor).
   690→   690→  let cmd;
   691→   691→  if (S.cruise) {
   692→   692→    if (performance.now() > S.cruise.until || Math.abs(controls.stickX) > 0.15 || Math.abs(controls.stickY) > 0.15) {
   693→   693→      if (S.cruise.until && performance.now() > S.cruise.until) ui.toast('Autofahrt beendet');
   694→   694→      else ui.toast('Autofahrt beendet (Stick-Eingriff)');
   695→   695→      S.cruise = null;
   696→   696→    }
   697→   697→  }
   698→   698→  if (S.cruise) {
   699→   699→    cmd = { vx: S.cruise.vx, yaw: S.cruise.yaw, climb: 0 };
   700→   700→  } else {
   701→   701→    cmd = controls.command(cfg);
   702→   702→  }
   703→   703→  const map = {};
   704→   704→  S.gait.step(sim, dtCtrl, cmd, map);
   705→   705→  for (const name in map) {
   706→   706→    const a = sim.actByName[name];
   707→   707→    if (a !== undefined) sim.ctrl[a] = map[name];
   708→   708→  }
   709→   709→  // Physik wirklich weiterschalten (Unterschritte je Regelrate)
   710→   710→  const substeps = Math.max(1, Math.round(CTRL_DT / sim.timestep));
   711→   711→  sim.stepN(substeps);
   712→   712→}

   714→   714→function policyCtrlStep() {
   715→   715→  const sim = S.sim, task = S.task, trainer = S.trainer;
   716→   716→  if (!trainer) return;
   717→   717→  const substeps = Math.max(1, Math.round((S.sim.cfg.ctrlDt || CTRL_DT) / sim.timestep));
   718→   718→  // Joystick-Steuerung (v2.5.0): bei ctrlMode 'joy' liefert der Stick die
   719→   719→  // Kommandos (vx = Vorwärts, yaw = Gieren), die die Policy im Training
   720→   720→  // mit Zufalls-Kommandos kennengelernt hat.
   721→   721→  if (task.kind === 'motion' && task.ctrlMode === 'joy') {
   722→   722→    const c = controls.command(S.sim.cfg);
   723→   723→    task.cmd.vx = c.vx; task.cmd.wz = c.yaw;
   724→   724→  }
   725→   725→  task.observe(sim, S.obsBuf);
   726→   726→  if (!finiteArr(S.obsBuf)) { sim.reset(); if (S.gait && S.gait.ph !== undefined) S.gait.ph = 0; return; }
   727→   727→  trainer.actDeterministic(S.obsBuf, S.actBuf);
   728→   728→  task.actionToCtrl(sim, S.actBuf);
   729→   729→  for (let i = 0; i < S.actBuf.length; i++) task.lastAct[i] = S.actBuf[i];
   730→   730→  sim.stepN(substeps);
   731→   731→  if (task.kind === 'motion') task.advance(CTRL_DT);
   732→   732→}

   744→   744→let lastFallLog = 0;
   745→   745→function checkFall() {
   746→   746→  const sim = S.sim;
   747→   747→  if (!sim) return;
   748→   748→  const o = 4 * sim.baseBody;
   749→   749→  const x = sim._xquat[o + 1], y = sim._xquat[o + 2]; // (x,y) der Quaternion [w,x,y,z]
   750→   750→  const upz = 1 - 2 * (x * x + y * y);
   751→   751→  const height = sim._xpos[3 * sim.baseBody + 2];
   752→   752→  const limit = sim.cfg.drone ? 0.35 : 0.32;
   753→   753→  const zMin = sim.cfg.drone ? 0.05 : sim.cfg.done.zMin * 0.8;
   754→   754→  if (upz < limit || height < zMin) {
   755→   755→    const now = performance.now();
   756→   756→    if (now - lastFallLog > 2500) {
   757→   757→      log('Sturz erkannt — Auto-Reset', 'warn');
   758→   758→      lastFallLog = now;
   759→   759→    }
   760→   760→    sim.reset();
   761→   761→    if (S.gait && S.gait.ph !== undefined) S.gait.ph = 0;
   762→   762→    S.epReward = 0;
   763→   763→  }
   764→   764→}

   831→   831→// ── Master-Loop ─────────────────────────────────────────────
   832→   832→let lastT = 0, statusT = 0, chartT = 0;
   833→   833→function loop(now) {
   834→   834→  requestAnimationFrame(loop);
   835→   835→  const dtRaw = (now - lastT) / 1000;
   836→   836→  lastT = now;
   837→   837→  const dt = Math.min(Math.max(dtRaw, 0.0001), 0.05);
   838→   838→  const fps = 1 / dtRaw;
   839→   839→
   840→   840→  if (!S.sim) return;
   841→   841→  controls.tick(dt, r3d);
   842→   842→  if (controls.consumeReset()) resetRobot();
   843→   843→
   844→   844→  if (S.training) {
   845→   845→    // Trainingsbetrieb: Zeitbudget pro Frame
   846→   846→    const budget = S.speedMode === 'max' ? 12e3 : 0; // µs
   847→   847→    const t0 = performance.now();
   848→   848→    const nSteps = S.speedMode === 'max' ? 1e9 : (S.speedMode === '16' ? 16 : S.speedMode === '4' ? 4 : 1);
   849→   849→    let done = 0;
   850→   850→    while (done < nSteps) {
   851→   851→      trainCtrlStep();
   852→   852→      done++;
   853→   853→      if (budget && performance.now() - t0 > budget) break;
   854→   854→    }
   855→   855→    const el = performance.now() - t0;
   856→   856→    S._stepTimes.push({ n: done, ms: el });
   857→   857→    if (S._stepTimes.length > 30) S._stepTimes.shift();
   858→   858→    let sn = 0, sm = 0;
   859→   859→    for (const s of S._stepTimes) { sn += s.n; sm += s.ms; }
   860→   860→    S.stepsPerSec = sm > 0 ? (sn / sm) * 1000 : 0;
   861→   861→  } else {
   862→   862→    // Echtzeit: MANUELL oder POLICY
   863→   863→    const cdt = S.sim.cfg.ctrlDt || CTRL_DT;
   864→   864→    S.acc += dt;
   865→   865→    let guard = 0;
   866→   866→    while (S.acc >= cdt && guard < 10) {
   867→   867→      S.acc -= cdt;
   868→   868→      guard++;
   869→   869→      if (S.mode === 'policy' && S.trainer) policyCtrlStep();
   870→   870→      else {
   871→   871→        applyGait(cdt);
   872→   872→        // Lehrer läuft auch im MANUELL-Modus weiter (Vorschau der Referenz)
   873→   873→        if (S.task && S.task.kind === 'motion') S.task.advance(cdt);
   874→   874→      }
   875→   875→      checkFall();
   876→   876→    }
   877→   877→    S.stepsPerSec = Math.max(1, Math.round(CTRL_DT)) * 0 + fps * Math.max(1, Math.round(CTRL_DT / S.sim.timestep));
   878→   878→  }
   879→   879→
   880→   880→  r3d.updateFrame(S.sim, dt);
   881→   881→  // Geist: Referenzpose mitlaufen lassen — Lehrer (Original) + G1-Geist
   882→   882→  // folgen BEIDE der Root-Bahn (der Lehrer läuft wirklich durchs Feld,
   883→   883→  // der Roboter lernt, ihm zu folgen — nicht „auf der Stelle" zu gehen)
   884→   884→  if (S.ghostOn && S.task && S.task.kind === 'motion' && (r3d.ghostGroups || r3d.sourceGhost)) {
   885→   885→    const clip = S.task.clip;
   886→   886→    const fr = Math.floor(S.task.phase * clip.n) % clip.n;
   887→   887→    // Loop-Rebase: Referenz läuft nach jedem Durchlauf WEITER (kein Teleport)
   888→   888→    const rr = clip.root && clip.yaw && S.task.refRoot ? S.task.refRoot(S.task.phase, [0, 0, 0]) : null;
   889→   889→    if (r3d.sourceGhost) {
   890→   890→      r3d.updateSourceGhost(fr);
   891→   891→      if (rr) r3d.setSourceGhostLoop(rr[0] - clip.root[2 * fr], rr[1] - clip.root[2 * fr + 1]);
   892→   892→    }
   893→   893→    if (r3d.ghostGroups) {
   894→   894→      const gh = S.sim.makeGhostData();
   895→   895→      // baseQ: Lehrer-Nick/Roll (z. B. Zombie-Beuge) — der G1-Geist nimmt die
   896→   896→      // ABSOLUTE Lehrer-Pose an statt aufrecht daneben zu stehen
   897→   897→      const bq = clip.baseQ ? clip.baseQ.subarray(4 * fr, 4 * fr + 4) : null;
   898→   898→      if (rr) {
   899→   899→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], rr[0], rr[1], rr[2], bq);
   900→   900→      } else if (clip.root && clip.yaw) {
   901→   901→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], clip.root[2 * fr], clip.root[2 * fr + 1], clip.yaw[fr], bq);
   902→   902→      } else {
   903→   903→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], 0, 0, 0, bq);
   904→   904→      }
   905→   905→      r3d.updateGhost(gh);
   906→   906→    }
   907→   907→  }
   908→   908→  r3d.render();
   909→   909→
   910→   910→  // Statuszeile (5 Hz)
   911→   911→  statusT += dt;
   912→   912→  if (statusT > 0.2) {
   913→   913→    statusT = 0;
   914→   914→    ui.status(S.sim.baseSpeed(), S.sim.baseHeight(), fps);
   915→   915→    if (ui.$('trainSheet') && !ui.$('trainSheet').classList.contains('hidden')) {
   916→   916→      const m = S.trainer ? S.trainer : null;
   917→   917→      ui.trainStats({
   918→   918→        reward: S.trainer ? (lastEma().toFixed(2).replace('.', ',')) : '–',
   919→   919→        episodes: S.episodes,
   920→   920→        steps: S.trainer ? S.trainer.stepCount : 0,
   921→   921→        rate: S.training ? S.stepsPerSec : 0,
   922→   922→      });
   923→   923→    }
   924→   924→  }
   925→   925→  chartT += dt;
   926→   926→  if (chartT > 0.35) {
   927→   927→    chartT = 0;
   928→   928→    if (!ui.$('trainSheet').classList.contains('hidden')) ui.drawChart();
   929→   929→  }
   930→   930→}

  1180→  1180→async function runBC() {
  1181→  1181→  if (_bcRunning) return;

  1201→  1201→      const loss = trainer.bcEpoch(ds.X, ds.Y, ds.n, 128);

  1204→  1204→        ui.$('glbStatus').textContent = 'BC Epoche ' + (ep + 1) + '/' + EPOCHS + ' · MSE ' + loss.toFixed(4);
  1205→  1205→        await new Promise((r) => setTimeout(r, 0));
  1206→  1206→      }
  1207→  1207→    }
  1208→  1208→    ui.toast('BC fertig — jetzt PPO verfeinern');

  1214→  1214→    ui.toast('BC fehlgeschlagen: ' + err.message, true);
  1215→  1215→  } finally {
  1216→  1216→    _bcRunning = false;
  1217→  1217→  }
