    66→    66→async function loadRobot(id, first = false) {
    67→    67→  if (S.switching) return;
    68→    68→  const cfg = getRobot(id);
    69→    69→  if (!cfg) return;
    70→    70→  S.switching = true;
    71→    71→  try {
    72→    72→    stopTraining(true);
    73→    73→    if (!first) ui.setRobotLoading(id);
    74→    74→    await new Promise(r => setTimeout(r, first ? 0 : 30)); // Frame zum Zeichnen
    75→    75→
    76→    76→    log(`Kompiliere ${cfg.dir} (${cfg.longName}) …`);
    77→    77→    const t0 = performance.now();
    78→    78→    await fetchModelIntoFS('models/' + cfg.dir);
    79→    79→    const sim = new RobotSim(cfg, cfg.scene);
    80→    80→    const ms = Math.round(performance.now() - t0);
    81→    81→
    82→    82→    // Alte Simulation freigeben
    83→    83→    if (S.sim) {
    84→    84→      const oldDir = S.sim.cfg.dir;
    85→    85→      S.sim.dispose();
    86→    86→      if (oldDir !== cfg.dir) removeModelFromFS(oldDir);
    87→    87→      S.sim = null;
    88→    88→    }
    89→    89→    S.sim = sim;
    90→    90→    S.robotId = id;
    91→    91→
    92→    92→    // Gang-/Flugregler
    93→    93→    S.gait = cfg.gait(cfg);
    94→    94→    if (cfg.drone && S.gait.init) { S.gait.init(sim); sim.flightCtl = S.gait; }
    95→    95→    // Trainingsaufgabe: GLB-Motion-Tracking (nur G1 mit aktivem Clip) sonst Geschwindigkeit
    96→    96→    if (id === 'g1' && S.motionClip) {
    97→    97→      S.task = makeMotionTask(cfg, S.motionClip, sim);
    98→    98→    } else {
    99→    99→      S.task = cfg.task(cfg);
   100→   100→    }
   101→   101→    S.task.reset(new RNG(4242), sim);
   102→   102→    S.obsBuf = new Float32Array(S.task.obsDim);
   103→   103→    S.actBuf = new Float32Array(S.task.actDim);
   104→   104→    ui.$('glbSection').classList.toggle('hidden', id !== 'g1');

   106→   106→    applySavedAICfg(id);
   107→   107→
   108→   108→    // Gespeicherte Policy für DIESE Aufgabenart laden (falls vorhanden)
   109→   109→    S.trainer = null;
   110→   110→    const saved = loadPolicy(id);
   111→   111→    if (saved) { S.trainer = saved; }
   112→   112→
   113→   113→    // Welt & Optik
   114→   114→    r3d.buildFromModel(sim);
   115→   115→    r3d.camDist = cfg.dist; controls._camDist = cfg.dist;
   116→   116→    sim.reset();
   117→   117→    S.epReward = 0; S.episodes = 0;
   118→   118→
   119→   119→    // UI
   120→   120→    ui.setRobotActive(id);
   121→   121→    ui.setRobotTitle(cfg);
   122→   122→    ui.setDroneMode(!!cfg.drone);
   123→   123→    ui.policyAvailable(!!S.trainer);
   124→   124→    ui.setMode(S.mode);
   125→   125→    if (S.mode === 'policy' && !S.trainer) { S.mode = 'manuell'; ui.setMode(S.mode); }
   126→   126→
   127→   127→    const meshes = countMeshes(sim);
   128→   128→    log(`MJCF OK: ${sim.nu} Aktuatoren · qpos ${sim.nq} · dof ${sim.nv} · ${meshes} Mesh-Geoms · dt ${sim.timestep}s (${ms} ms)`, 'ok');

   185→   185→function aiCtx() {
   186→   186→  const cfg = S.sim ? S.sim.cfg : null;
   187→   187→  const kind = aiTaskKind();
   188→   188→  const cur = {};
   189→   189→  if (kind === 'speed' && cfg) {
   190→   190→    cur.rW = cfg.rW; cur.cmd = cfg.cmd; cur.done = cfg.done;
   191→   191→    cur.actSpan = cfg.actSpan; cur.speedMax = cfg.speedMax;
   192→   192→  } else if (kind === 'motion') {
   193→   193→    cur.motionR = MOTION_R;
   194→   194→  } else if (kind === 'hover' && cfg) {
   195→   195→    cur.hoverR = HOVER_R; cur.cmd = cfg.cmd;
   196→   196→  }
   197→   197→  cur.ppo = S.trainer
   198→   198→    ? (({ lr, gamma, lam, clip, epochs, mb, T, cV, cE, maxGrad }) => ({ lr, gamma, lam, clip, epochs, mb, T, cV, cE, maxGrad }))(S.trainer.h)
   199→   199→    : Object.assign({}, DEFAULT_PPO, PPO_OVERRIDES);
   200→   200→  return {
   201→   201→    robot: S.robotId,
   202→   202→    robotName: cfg ? cfg.longName : '—',
   203→   203→    taskKind: kind,
   204→   204→    current: cur,
   205→   205→    history: S.aiHistory,
   206→   206→  };
   207→   207→}

   209→   209→function _aiPPOApply(ppo, touched) {
   210→   210→  for (const [k, v] of Object.entries(ppo)) {
   211→   211→    PPO_OVERRIDES[k] = v;
   212→   212→    if (S.trainer && k !== 'T') S.trainer.h[k] = v;
   213→   213→    touched.push('ppo.' + k);
   214→   214→  }

   217→   217→// Patch auf die laufende Konfiguration anwenden (cfg-Objekte werden von
   218→   218→// den Aufgaben live gelesen → Wirkung ohne Neustart).
   219→   219→function applyAIPatch(patch, opts = {}) {
   220→   220→  if (!patch || !Object.keys(patch).length) return false;
   221→   221→  const cfg = S.sim ? S.sim.cfg : null;
   222→   222→  const kind = aiTaskKind();
   223→   223→  const touched = [];
   224→   224→  if (patch.push && patch.push.impulse !== undefined) {
   225→   225→    S.pushStrength = savePushStrength(patch.push.impulse);
   226→   226→    touched.push('push.impulse=' + S.pushStrength.toFixed(1));
   227→   227→  }
   228→   228→  if (cfg) {
   229→   229→    if (kind === 'speed') {
   230→   230→      if (patch.rW) for (const [k, v] of Object.entries(patch.rW)) { cfg.rW[k] = v; touched.push('rW.' + k); }
   231→   231→      if (patch.cmd) {
   232→   232→        if (patch.cmd.vx) { cfg.cmd.vx = patch.cmd.vx.slice(); touched.push('cmd.vx=' + patch.cmd.vx.map(x => x.toFixed(2)).join('..')); }
   233→   233→        if (patch.cmd.yaw) { cfg.cmd.yaw = patch.cmd.yaw.slice(); touched.push('cmd.yaw'); }
   234→   234→      }
   235→   235→      if (patch.done) for (const [k, v] of Object.entries(patch.done)) { cfg.done[k] = v; touched.push('done.' + k); }
   236→   236→      if (patch.actSpan !== undefined) { cfg.actSpan = patch.actSpan; touched.push('actSpan'); }
   237→   237→    } else if (kind === 'hover') {
   238→   238→      if (patch.hoverR) for (const [k, v] of Object.entries(patch.hoverR)) { HOVER_R[k] = v; touched.push('hoverR.' + k); }
   239→   239→      if (patch.cmd && patch.cmd.vx) { cfg.cmd.vx = patch.cmd.vx.slice(); touched.push('cmd.vx'); }
   240→   240→    }
   241→   241→  }
   242→   242→  if (patch.motionR) for (const [k, v] of Object.entries(patch.motionR)) { MOTION_R[k] = v; touched.push('motionR.' + k); }
   243→   243→  if (patch.hoverR && kind !== 'hover') for (const [k, v] of Object.entries(patch.hoverR)) { HOVER_R[k] = v; touched.push('hoverR.' + k); }
   244→   244→  if (patch.ppo) _aiPPOApply(patch.ppo, touched);
   245→   245→  saveAICfg();
   246→   246→  log('KI-Anpassung übernommen: ' + touched.join(', '), 'ok');
   247→   247→  ui.toast('KI-Anpassung übernommen' + (opts.resetTraining ? ' — Training zurückgesetzt' : ''));
   248→   248→  if (opts.resetTraining) {
   249→   249→    stopTraining(true);
   250→   250→    S.trainer = null;
   251→   251→    ui.resetRewards();
   252→   252→    S.episodes = 0;
   253→   253→    if (S.sim) S.sim.reset();
   254→   254→    ui.$('tStart').textContent = 'Training starten';
   255→   255→    ui.$('tStart').classList.remove('btn-stop');
   256→   256→    ui.trainStats({ reward: '–', episodes: 0, steps: 0, rate: 0 });
   257→   257→    ui.drawChart();
   258→   258→    log('Training zurückgesetzt (KI-Anpassung war groß — Netz lernt neu)', 'warn');
   259→   259→  }
   260→   260→  return true;
   261→   261→}

   303→   303→function doPush(dir = 'auto', strength = null) {
   304→   304→  if (!S.sim) return 'Kein Roboter geladen';
   305→   305→  const st = strength != null ? Math.min(4, Math.max(0.2, strength)) : S.pushStrength;
   306→   306→  S.sim.pushRandom(dir, st);
   307→   307→  controls.buzz(30);
   308→   308→  ui.toast('SCHUBS! (' + dir + ', Stärke ' + st.toFixed(1) + ')');
   309→   309→  log('Schubs: dir=' + dir + ' Stärke=' + st.toFixed(1) + ' (Δv≈' + (st * 12 / (S.sim._totalMass || 20)).toFixed(2) + ' m/s)', 'warn');
   310→   310→  return 'Schubs ausgeführt (dir=' + dir + ', strength=' + st.toFixed(1) + ')';
   311→   311→}

   313→   313→function observeState() {
   314→   314→  const sim = S.sim;
   315→   315→  let up = null;
   316→   316→  if (sim) {
   317→   317→    const o = 4 * sim.baseBody;
   318→   318→    const x = sim._xquat[o + 1], y = sim._xquat[o + 2];
   319→   319→    up = 1 - 2 * (x * x + y * y);
   320→   320→  }
   321→   321→  return JSON.stringify({
   322→   322→    robot: S.robotId,
   323→   323→    mode: S.mode,
   324→   324→    training: S.training,
   325→   325→    speed: sim ? +sim.baseSpeed().toFixed(2) : null,
   326→   326→    height: sim ? +sim.baseHeight().toFixed(2) : null,
   327→   327→    upright: up !== null ? +up.toFixed(2) : null,
   328→   328→    episodes: S.episodes,
   329→   329→    policySaved: !!S.trainer,
   330→   330→    motionClips: S.clips.map((c, i) => ({ index: i, name: c.name })),
   331→   331→    activeClip: S.motionClip ? S.motionClip.name : null,
   332→   332→    pushStrength: S.pushStrength,
   333→   333→    joystick: controls.joyMap,
   334→   334→    buttons: S.aiButtons.map(b => ({ id: b.id, label: b.label, action: b.action })),
   335→   335→  });

   338→   338→function executeAction(action, depth = 0) {
   339→   339→  const act = action;
   340→   340→  if (!act || typeof act !== 'object') return 'Ungültige Aktion';
   341→   341→  switch (act.type) {
   342→   342→    case 'reset':
   343→   343→      resetRobot();
   344→   344→      return 'Roboter zurückgesetzt';
   345→   345→    case 'push':
   346→   346→      return doPush(act.dir || 'auto', act.strength != null ? act.strength : null);
   347→   347→    case 'cmd': {
   348→   348→      if (!S.sim) return 'Kein Roboter geladen';
   349→   349→      S.cruise = { vx: act.vx, yaw: act.yaw, until: performance.now() + act.ms };
   350→   350→      return 'Autofahrt gestartet: vx=' + act.vx.toFixed(2) + ' m/s, yaw=' + act.yaw.toFixed(2) + ' rad/s für ' + Math.round(act.ms / 1000) + ' s (endet früher bei Stick-Bewegung)';
   351→   351→    }
   352→   352→    case 'mode': {
   353→   353→      if (act.mode === 'policy' && !S.trainer) return 'Keine Policy vorhanden — erst Training starten';
   354→   354→      S.mode = act.mode;
   355→   355→      ui.setMode(S.mode);
   356→   356→      if (S.sim) S.sim.reset();
   357→   357→      return 'Modus: ' + act.mode.toUpperCase();
   358→   358→    }
   359→   359→    case 'clip': {
   360→   360→      const rec = S.clips[act.index];
   361→   361→      if (!rec) return 'Kein Clip an Index ' + act.index + ' (vorhanden: ' + S.clips.length + ')';
   362→   362→      activateClip(rec).catch(err => log('Clip-Aktivierung fehlgeschlagen: ' + err.message, 'err'));
   363→   363→      return 'Clip-Aktivierung gestartet: ' + (rec.name || act.index);
   364→   364→    }
   365→   365→    case 'macro': {
   366→   366→      if (depth > 2) return 'Makro-Verschachtelung zu tief';
   367→   367→      let t = 0;
   368→   368→      const results = [];
   369→   369→      for (const step of act.steps.slice(0, 6)) {
   370→   370→        if (step.waitMs !== undefined) { t += step.waitMs; continue; }
   371→   371→        const wait = t;
   372→   372→        const sub = step;
   373→   373→        setTimeout(() => executeAction(sub, depth + 1), wait);
   374→   374→        results.push(sub.type + '@' + wait + 'ms');
   375→   375→      }
   376→   376→      return 'Makro geplant: ' + (results.join(', ') || 'nur Wartezeiten') + ' (Gesamt ' + t + ' ms)';
   377→   377→    }
   378→   378→    default:
   379→   379→      return 'Unbekannter Aktionstyp: ' + (act.type || '?');
   380→   380→  }
   381→   381→}

   384→   384→function execTool(tool, args) {
   385→   385→  try {
   386→   386→    if (tool === 'observe') return observeState();
   387→   387→    if (tool === 'applyConfig') {
   388→   388→      const ok = applyAIPatch(args.patch, { resetTraining: args.resetTraining });
   389→   389→      return ok ? 'Konfiguration angewendet: ' + JSON.stringify(args.patch) : 'Patch war leer — nichts geändert';
   390→   390→    }
   391→   391→    if (tool === 'addButton') {
   392→   392→      const r = addButton({ label: args.label, action: args.action });
   393→   393→      if (!r.ok) return 'Fehler: ' + r.reason;
   394→   394→      S.aiButtons = r.list;
   395→   395→      renderAIButtons();
   396→   396→      return 'Button erstellt: id=' + r.button.id + ' label="' + r.button.label + '" action=' + JSON.stringify(r.button.action);
   397→   397→    }
   398→   398→    if (tool === 'removeButton') {
   399→   399→      const r = removeButton(args.id);
   400→   400→      if (!r.ok) return 'Fehler: ' + r.reason;
   401→   401→      S.aiButtons = r.list;
   402→   402→      renderAIButtons();
   403→   403→      return 'Button entfernt: ' + args.id;
   404→   404→    }
   405→   405→    if (tool === 'mapJoystick') {
   406→   406→      controls.joyMap = validateJoyMap(args);
   407→   407→      saveJoyMap(controls.joyMap);
   408→   408→      return 'Joystick-Map gesetzt: ' + JSON.stringify(controls.joyMap);
   409→   409→    }
   410→   410→    return 'Unbekanntes Werkzeug: ' + tool;
   411→   411→  } catch (e) {
   412→   412→    return 'Werkzeug-Fehler: ' + e.message;
   413→   413→  }

   638→   638→// Ein Trainingsschritt (auf Regelrate CTRL_DT)
   639→   639→function trainCtrlStep() {
   640→   640→  const sim = S.sim, task = S.task, trainer = S.trainer;
   641→   641→  const substeps = Math.max(1, Math.round(CTRL_DT / sim.timestep));
   642→   642→  if (task.stepsLeft <= 0) task.sampleCmd(trainer.rng);
   643→   643→
   644→   644→  const o = task.observe(sim, S.obsBuf);
   645→   645→  if (!finiteArr(S.obsBuf)) { sim.reset(); task.reset(trainer.rng, sim); return; }
   646→   646→  trainer.norm.update(S.obsBuf);
   647→   647→  const { act, logp, value } = trainer.act(S.obsBuf, false);
   648→   648→  task.actionToCtrl(sim, act);
   649→   649→  sim.stepN(substeps);
   650→   650→  if (task.kind === 'motion') task.advance(CTRL_DT);
   651→   651→  const { r, done } = task.reward(sim);
   652→   652→  for (let i = 0; i < act.length; i++) task.lastAct[i] = act[i];
   653→   653→  S.epReward += r;
   654→   654→
   655→   655→  const full = trainer.store(S.obsBuf, act, logp, r, done, value);
   656→   656→  if (done) {
   657→   657→    S.episodes++;
   658→   658→    ui.pushEpisodeReward(S.epReward);
   659→   659→    S.epReward = 0;
   660→   660→    sim.reset();
   661→   661→    task.reset(trainer.rng, sim);
   662→   662→  }
   663→   663→  if (full) {
   664→   664→    const lastObs = task.observe(sim, S.obsBuf);
   665→   665→    const lastVal = trainer.act(S.obsBuf, true).value;
   666→   666→    const m = trainer.finishAndUpdate(lastVal);
   667→   667→    m._lastMetrics = m;
   668→   668→  }
   669→   669→}

   672→   672→function applyGait(dtCtrl) {
   673→   673→  const sim = S.sim, cfg = S.sim.cfg;
   674→   674→  dtCtrl = cfg.ctrlDt || dtCtrl;
   675→   675→  // KI-Autofahrt (Cruise): fester Fahrbefehl, endet an Zeitgrenze oder
   676→   676→  // sobald der Nutzer den Stick bewegt (Eingriff geht immer vor).
   677→   677→  let cmd;
   678→   678→  if (S.cruise) {
   679→   679→    if (performance.now() > S.cruise.until || Math.abs(controls.stickX) > 0.15 || Math.abs(controls.stickY) > 0.15) {
   680→   680→      if (S.cruise.until && performance.now() > S.cruise.until) ui.toast('Autofahrt beendet');
   681→   681→      else ui.toast('Autofahrt beendet (Stick-Eingriff)');
   682→   682→      S.cruise = null;
   683→   683→    }
   684→   684→  }
   685→   685→  if (S.cruise) {
   686→   686→    cmd = { vx: S.cruise.vx, yaw: S.cruise.yaw, climb: 0 };
   687→   687→  } else {
   688→   688→    cmd = controls.command(cfg);
   689→   689→  }
   690→   690→  const map = {};
   691→   691→  S.gait.step(sim, dtCtrl, cmd, map);
   692→   692→  for (const name in map) {
   693→   693→    const a = sim.actByName[name];
   694→   694→    if (a !== undefined) sim.ctrl[a] = map[name];
   695→   695→  }
   696→   696→  // Physik wirklich weiterschalten (Unterschritte je Regelrate)
   697→   697→  const substeps = Math.max(1, Math.round(CTRL_DT / sim.timestep));
   698→   698→  sim.stepN(substeps);
   699→   699→}

   701→   701→function policyCtrlStep() {
   702→   702→  const sim = S.sim, task = S.task, trainer = S.trainer;
   703→   703→  if (!trainer) return;
   704→   704→  const substeps = Math.max(1, Math.round((S.sim.cfg.ctrlDt || CTRL_DT) / sim.timestep));

   707→   707→  trainer.actDeterministic(S.obsBuf, S.actBuf);
   708→   708→  task.actionToCtrl(sim, S.actBuf);
   709→   709→  for (let i = 0; i < S.actBuf.length; i++) task.lastAct[i] = S.actBuf[i];
   710→   710→  sim.stepN(substeps);
   711→   711→  if (task.kind === 'motion') task.advance(CTRL_DT);

   724→   724→let lastFallLog = 0;
   725→   725→function checkFall() {
   726→   726→  const sim = S.sim;
   727→   727→  if (!sim) return;
   728→   728→  const o = 4 * sim.baseBody;
   729→   729→  const x = sim._xquat[o + 1], y = sim._xquat[o + 2]; // (x,y) der Quaternion [w,x,y,z]
   730→   730→  const upz = 1 - 2 * (x * x + y * y);
   731→   731→  const height = sim._xpos[3 * sim.baseBody + 2];
   732→   732→  const limit = sim.cfg.drone ? 0.35 : 0.32;
   733→   733→  const zMin = sim.cfg.drone ? 0.05 : sim.cfg.done.zMin * 0.8;
   734→   734→  if (upz < limit || height < zMin) {
   735→   735→    const now = performance.now();
   736→   736→    if (now - lastFallLog > 2500) {
   737→   737→      log('Sturz erkannt — Auto-Reset', 'warn');
   738→   738→      lastFallLog = now;
   739→   739→    }
   740→   740→    sim.reset();
   741→   741→    if (S.gait && S.gait.ph !== undefined) S.gait.ph = 0;
   742→   742→    S.epReward = 0;
   743→   743→  }
   744→   744→}

   807→   807→// ── Master-Loop ─────────────────────────────────────────────
   808→   808→let lastT = 0, statusT = 0, chartT = 0;
   809→   809→function loop(now) {
   810→   810→  requestAnimationFrame(loop);
   811→   811→  const dtRaw = (now - lastT) / 1000;
   812→   812→  lastT = now;
   813→   813→  const dt = Math.min(Math.max(dtRaw, 0.0001), 0.05);
   814→   814→  const fps = 1 / dtRaw;
   815→   815→
   816→   816→  if (!S.sim) return;
   817→   817→  controls.tick(dt, r3d);
   818→   818→  if (controls.consumeReset()) resetRobot();
   819→   819→
   820→   820→  if (S.training) {
   821→   821→    // Trainingsbetrieb: Zeitbudget pro Frame
   822→   822→    const budget = S.speedMode === 'max' ? 12e3 : 0; // µs
   823→   823→    const t0 = performance.now();
   824→   824→    const nSteps = S.speedMode === 'max' ? 1e9 : (S.speedMode === '16' ? 16 : S.speedMode === '4' ? 4 : 1);
   825→   825→    let done = 0;
   826→   826→    while (done < nSteps) {
   827→   827→      trainCtrlStep();
   828→   828→      done++;
   829→   829→      if (budget && performance.now() - t0 > budget) break;
   830→   830→    }
   831→   831→    const el = performance.now() - t0;
   832→   832→    S._stepTimes.push({ n: done, ms: el });
   833→   833→    if (S._stepTimes.length > 30) S._stepTimes.shift();
   834→   834→    let sn = 0, sm = 0;
   835→   835→    for (const s of S._stepTimes) { sn += s.n; sm += s.ms; }
   836→   836→    S.stepsPerSec = sm > 0 ? (sn / sm) * 1000 : 0;
   837→   837→  } else {
   838→   838→    // Echtzeit: MANUELL oder POLICY
   839→   839→    const cdt = S.sim.cfg.ctrlDt || CTRL_DT;
   840→   840→    S.acc += dt;
   841→   841→    let guard = 0;
   842→   842→    while (S.acc >= cdt && guard < 10) {
   843→   843→      S.acc -= cdt;
   844→   844→      guard++;
   845→   845→      if (S.mode === 'policy' && S.trainer) policyCtrlStep();
   846→   846→      else {
   847→   847→        applyGait(cdt);
   848→   848→        // Lehrer läuft auch im MANUELL-Modus weiter (Vorschau der Referenz)
   849→   849→        if (S.task && S.task.kind === 'motion') S.task.advance(cdt);
   850→   850→      }
   851→   851→      checkFall();
   852→   852→    }
   853→   853→    S.stepsPerSec = Math.max(1, Math.round(CTRL_DT)) * 0 + fps * Math.max(1, Math.round(CTRL_DT / S.sim.timestep));
   854→   854→  }
   855→   855→
   856→   856→  r3d.updateFrame(S.sim, dt);
   857→   857→  // Geist: Referenzpose mitlaufen lassen — Lehrer (Original) + G1-Geist
   858→   858→  // folgen BEIDE der Root-Bahn (der Lehrer läuft wirklich durchs Feld,
   859→   859→  // der Roboter lernt, ihm zu folgen — nicht „auf der Stelle" zu gehen)
   860→   860→  if (S.ghostOn && S.task && S.task.kind === 'motion' && (r3d.ghostGroups || r3d.sourceGhost)) {
   861→   861→    const clip = S.task.clip;
   862→   862→    const fr = Math.floor(S.task.phase * clip.n) % clip.n;
   863→   863→    // Loop-Rebase: Referenz läuft nach jedem Durchlauf WEITER (kein Teleport)
   864→   864→    const rr = clip.root && clip.yaw && S.task.refRoot ? S.task.refRoot(S.task.phase, [0, 0, 0]) : null;
   865→   865→    if (r3d.sourceGhost) {
   866→   866→      r3d.updateSourceGhost(fr);
   867→   867→      if (rr) r3d.setSourceGhostLoop(rr[0] - clip.root[2 * fr], rr[1] - clip.root[2 * fr + 1]);
   868→   868→    }
   869→   869→    if (r3d.ghostGroups) {
   870→   870→      const gh = S.sim.makeGhostData();
   871→   871→      // baseQ: Lehrer-Nick/Roll (z. B. Zombie-Beuge) — der G1-Geist nimmt die
   872→   872→      // ABSOLUTE Lehrer-Pose an statt aufrecht daneben zu stehen
   873→   873→      const bq = clip.baseQ ? clip.baseQ.subarray(4 * fr, 4 * fr + 4) : null;
   874→   874→      if (rr) {
   875→   875→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], rr[0], rr[1], rr[2], bq);
   876→   876→      } else if (clip.root && clip.yaw) {
   877→   877→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], clip.root[2 * fr], clip.root[2 * fr + 1], clip.yaw[fr], bq);
   878→   878→      } else {
   879→   879→        S.sim.setGhostPose(gh, clip.q, fr * clip.nu, clip.h[fr], 0, 0, 0, bq);
   880→   880→      }
   881→   881→      r3d.updateGhost(gh);
   882→   882→    }
   883→   883→  }
   884→   884→  r3d.render();
   885→   885→
   886→   886→  // Statuszeile (5 Hz)
   887→   887→  statusT += dt;
   888→   888→  if (statusT > 0.2) {
   889→   889→    statusT = 0;
   890→   890→    ui.status(S.sim.baseSpeed(), S.sim.baseHeight(), fps);
   891→   891→    if (ui.$('trainSheet') && !ui.$('trainSheet').classList.contains('hidden')) {
   892→   892→      const m = S.trainer ? S.trainer : null;
   893→   893→      ui.trainStats({
   894→   894→        reward: S.trainer ? (lastEma().toFixed(2).replace('.', ',')) : '–',
   895→   895→        episodes: S.episodes,
   896→   896→        steps: S.trainer ? S.trainer.stepCount : 0,
   897→   897→        rate: S.training ? S.stepsPerSec : 0,
   898→   898→      });
   899→   899→    }
   900→   900→  }
   901→   901→  chartT += dt;
   902→   902→  if (chartT > 0.35) {
   903→   903→    chartT = 0;
   904→   904→    if (!ui.$('trainSheet').classList.contains('hidden')) ui.drawChart();
   905→   905→  }
   906→   906→}

  1126→  1126→async function runBC() {
  1127→  1127→  if (_bcRunning) return;
  1128→  1128→  if (!S.task || S.task.kind !== 'motion') { ui.toast('Zuerst eine GLB-Referenz wählen', true); return; }

  1133→  1133→    await new Promise((r) => setTimeout(r, 30));
  1134→  1134→    const ds = task.buildBCDataset(S.sim);
  1135→  1135→    log('BC-Datensatz: ' + ds.n + ' Frames (Geist + Rauschen) — Etikett = nächste Referenzpose');

  1138→  1138→    }

  1140→  1140→    for (let f = 0; f < ds.n; f++) {
  1141→  1141→      const o = new Float32Array(task.obsDim);
  1142→  1142→      for (let i = 0; i < task.obsDim; i++) o[i] = ds.X[f * task.obsDim + i];
  1143→  1143→      trainer.norm.update(o);
  1144→  1144→    }
  1145→  1145→    const EPOCHS = 60;
  1146→  1146→    for (let ep = 0; ep < EPOCHS; ep++) {
  1147→  1147→      const loss = trainer.bcEpoch(ds.X, ds.Y, ds.n, 128);
  1148→  1148→      if (ep % 10 === 0 || ep === EPOCHS - 1) {
  1149→  1149→        log('BC Epoche ' + (ep + 1) + '/' + EPOCHS + ' — MSE ' + loss.toFixed(4));
  1150→  1150→        ui.$('glbStatus').textContent = 'BC Epoche ' + (ep + 1) + '/' + EPOCHS + ' · MSE ' + loss.toFixed(4);
  1151→  1151→        await new Promise((r) => setTimeout(r, 0));
  1152→  1152→      }
  1153→  1153→    }
  1154→  1154→    ui.toast('BC fertig — jetzt PPO verfeinern');
  1155→  1155→    log('BC abgeschlossen. Empfehlung: Training starten (PPO-Residual auf der CPU).', 'ok');
  1156→  1156→    ui.$('glbStatus').textContent = 'BC fertig. PPO verfeinern mit „Training starten".';
  1157→  1157→  } catch (err) {
  1158→  1158→    console.error(err);
  1159→  1159→    log('BC-Fehler: ' + err.message, 'err');
  1160→  1160→    ui.toast('BC fehlgeschlagen: ' + err.message, true);
  1161→  1161→  } finally {
  1162→  1162→    _bcRunning = false;
  1163→  1163→  }
  1164→  1164→}
